JSTACK=/opt/codeconnect/openjdk-17.0.18/bin/jstack

while true; do
  RUNNABLE=$($JSTACK 3772 2>/dev/null | grep -A1 "dispatcher\|PassThrough" | grep -c "RUNNABLE")
  WAITING=$($JSTACK 3772 2>/dev/null | grep -A1 "dispatcher\|PassThrough" | grep -c "WAITING\|TIMED_WAITING")
  BLOCKED=$($JSTACK 3772 2>/dev/null | grep -A1 "dispatcher\|PassThrough" | grep -c "BLOCKED")
  echo "$(date +%H:%M:%S) | Active: $RUNNABLE | Waiting: $WAITING | Blocked: $BLOCKED"
  sleep 3
done
