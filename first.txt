SELECT 
    tnt.UM_DOMAIN_NAME AS BANK,
    TO_CHAR(app.ORGANIZATION) AS ORGANIZATION,
    TO_CHAR(app.NAME) AS APPLICATION_NAME,
    map.CONSUMER_KEY,
    COUNT(tok.ACCESS_TOKEN) AS ACTIVE_TOKEN_COUNT
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
JOIN 
    COCT42_AM.AM_APPLICATION_KEY_MAPPING map ON apps.CONSUMER_KEY = map.CONSUMER_KEY
JOIN 
    COCT42_AM.AM_APPLICATION app ON map.APPLICATION_ID = app.APPLICATION_ID
JOIN 
    COCT42_USER.UM_TENANT tnt ON tok.TENANT_ID = tnt.UM_ID
WHERE 
    tok.TOKEN_STATE = 'ACTIVE' 
    -- Using a simplified numeric comparison for the date logic
    AND (tok.TIME_CREATED + tok.VALIDITY_PERIOD) > 
        ((CAST(SYS_EXTRACT_UTC(SYSTIMESTAMP) AS DATE) - TO_DATE('01-01-1970', 'DD-MM-YYYY')) * 86400000)
GROUP BY 
    tnt.UM_DOMAIN_NAME, 
    TO_CHAR(app.ORGANIZATION), 
    TO_CHAR(app.NAME), 
    map.CONSUMER_KEY
ORDER BY 
    ACTIVE_TOKEN_COUNT DESC;
