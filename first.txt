SELECT 
    COUNT(CASE WHEN (tok.TIME_CREATED + (tok.VALIDITY_PERIOD / 86400000)) < SYSTIMESTAMP THEN 1 END) AS EXPIRED_BUT_STILL_IN_DB,
    COUNT(CASE WHEN (tok.TIME_CREATED + (tok.VALIDITY_PERIOD / 86400000)) >= SYSTIMESTAMP THEN 1 END) AS TRULY_ACTIVE_NOW,
    COUNT(*) AS TOTAL_IN_TABLE
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
WHERE 
    apps.CONSUMER_KEY = '11245';
