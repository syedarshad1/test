# Login to ih-prd-imse.prod.local
ssh user@ih-prd-imse.prod.local

# 1. Check total TIME_WAIT socket count
netstat -an | grep TIME_WAIT | wc -l
# or
ss -an | grep TIME_WAIT | wc -l

# Expected: if > 5,000, you've found the smoking gun.

# 2. Check ephemeral port availability
ss -tan | grep ESTAB | wc -l    # Active connections
ss -tan | grep TIME_WAIT | wc -l  # Zombie connections

# 3. See the actual kernel limit
cat /proc/sys/net/ipv4/ip_local_port_range
# Typical: 32768 65535  (28,767 ports available)

# 4. Check netstat distribution
netstat -an | awk '{print $6}' | sort | uniq -c | sort -rn
# or
ss -an | awk '{print $2}' | sort | uniq -c | sort -rn
