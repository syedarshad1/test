import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.io.Console;
import java.util.*;
import java.util.regex.*;
import java.net.*;
import java.util.concurrent.*;

public class OracleConnectionTest {

    private static final String GREEN  = "\u001B[32m";
    private static final String RED    = "\u001B[31m";
    private static final String YELLOW = "\u001B[33m";
    private static final String CYAN   = "\u001B[36m";
    private static final String RESET  = "\u001B[0m";
    private static final String BOLD   = "\u001B[1m";
    private static final int CONNECTION_TIMEOUT_SECONDS = 10;

    private static String extractServiceName(String jdbcUrl) {
        Pattern p = Pattern.compile("service_name=([^\\)]+)", Pattern.CASE_INSENSITIVE);
        Matcher m = p.matcher(jdbcUrl);
        return m.find() ? m.group(1).trim() : "UNKNOWN_SERVICE";
    }

    private static List<String> extractHosts(String jdbcUrl) {
        List<String> hosts = new ArrayList<>();
        Pattern p = Pattern.compile("host=([^\\)]+)", Pattern.CASE_INSENSITIVE);
        Matcher m = p.matcher(jdbcUrl);
        while (m.find()) hosts.add(m.group(1).trim());
        return hosts;
    }

    private static String extractPort(String jdbcUrl, String host) {
        // Find port for specific host
        Pattern p = Pattern.compile("host=" + Pattern.quote(host) + "\\)\\(protocol=tcp\\)\\(port=([0-9]+)\\)", Pattern.CASE_INSENSITIVE);
        Matcher m = p.matcher(jdbcUrl);
        if (m.find()) return m.group(1);
        
        p = Pattern.compile("\\(port=([0-9]+)\\)", Pattern.CASE_INSENSITIVE);
        m = p.matcher(jdbcUrl);
        if (m.find()) return m.group(1);
        
        return "1521";
    }

    private static String extractServiceNameFromUrl(String jdbcUrl) {
        Pattern p = Pattern.compile("service_name=([^\\)]+)", Pattern.CASE_INSENSITIVE);
        Matcher m = p.matcher(jdbcUrl);
        return m.find() ? m.group(1).trim() : "II_RDY_SVC";
    }

    private static String buildSingleHostUrl(String host, String port, String serviceName) {
        return "jdbc:oracle:thin:@(description=(address=(host=" + host + ")(protocol=tcp)(port=" + port + "))(connect_data=(service_name=" + serviceName + ")))";
    }

    private static String resolveIP(String host) {
        try {
            InetAddress addr = InetAddress.getByName(host);
            return addr.getHostAddress();
        } catch (Exception e) {
            return "DNS_LOOKUP_FAILED";
        }
    }

    private static String getBackendNode(Connection conn) {
        try {
            java.sql.Statement stmt = conn.createStatement();
            java.sql.ResultSet rs = stmt.executeQuery(
                    "SELECT sys_context('USERENV','SERVER_HOST') AS server_host FROM dual");

            String backendNode = "UNKNOWN_NODE";
            if (rs.next()) {
                backendNode = rs.getString("server_host");
            }

            rs.close();
            stmt.close();
            return backendNode;
        } catch (SQLException e) {
            return "UNKNOWN_NODE";
        }
    }

    private static Connection connectWithTimeout(String url, String user, String pass, int timeoutSeconds) 
            throws TimeoutException, SQLException {
        ExecutorService executor = Executors.newSingleThreadExecutor();
        Future<Connection> future = executor.submit(() -> DriverManager.getConnection(url, user, pass));

        try {
            return future.get(timeoutSeconds, TimeUnit.SECONDS);
        } catch (TimeoutException e) {
            future.cancel(true);
            throw e;
        } catch (Exception e) {
            if (e.getCause() instanceof SQLException) {
                throw (SQLException) e.getCause();
            }
            throw new SQLException(e);
        } finally {
            executor.shutdownNow();
        }
    }

    public static void main(String[] args) {

        System.out.println(CYAN + "=== Oracle JDBC Diagnostic Tool ===" + RESET);
        System.out.println(CYAN + "Connection timeout: " + CONNECTION_TIMEOUT_SECONDS + " seconds per host" + RESET);

        String tnsPath = (args.length > 0)
                ? args[0]
                : Paths.get("tns.txt").toAbsolutePath().toString();

        System.out.println(CYAN + "Using file: " + tnsPath + RESET);

        List<String> lines;

        try {
            lines = Files.readAllLines(Paths.get(tnsPath));
        } catch (Exception e) {
            System.out.println(RED + "FAILED: Unable to read " + tnsPath + RESET);
            return;
        }

        lines.removeIf(line -> line.trim().isEmpty() || line.trim().startsWith("#"));

        if (lines.isEmpty()) {
            System.out.println(RED + "No JDBC URLs found." + RESET);
            return;
        }

        Console console = System.console();
        if (console == null) {
            System.out.println(RED + "No console available." + RESET);
            return;
        }

        int entryNumber = 1;
        int totalEntries = lines.size();

        // ---------------------------------------------------------
        // ONE LOOP — ASK CREDENTIALS → TEST EACH HOST → REPORT
        // ---------------------------------------------------------
        for (String url : lines) {

            String service = extractServiceName(url);

            System.out.println(CYAN + "\n==================================================" + RESET);
            System.out.println(CYAN + "Entry " + entryNumber + " of " + totalEntries + RESET);

            // Extract hosts from JDBC URL
            List<String> scanHosts = extractHosts(url);
            System.out.println(CYAN + "Configured SCAN hosts: " + scanHosts + RESET);

            // Ask credentials immediately
            System.out.println(CYAN + "--- Credentials for " + service + " ---" + RESET);
            String user = console.readLine("Enter username for " + service + ": ");
            char[] passChars = console.readPassword("Enter password for " + service + ": ");
            String pass = new String(passChars);

            String serviceName = extractServiceNameFromUrl(url);
            String connectedHost = null;
            String backendNode = null;
            long totalTime = 0;

            // Try each host individually
            for (String host : scanHosts) {
                String port = extractPort(url, host);
                String singleHostUrl = buildSingleHostUrl(host, port, serviceName);
                
                System.out.println(YELLOW + "Testing host: " + host + ":" + port + RESET);
                
                long start = System.currentTimeMillis();
                Connection conn = null;

                try {
                    // Connect with timeout
                    conn = connectWithTimeout(singleHostUrl, user, pass, CONNECTION_TIMEOUT_SECONDS);
                    totalTime = System.currentTimeMillis() - start;

                    // Connection successful! Get backend node
                    backendNode = getBackendNode(conn);
                    connectedHost = host;
                    
                    System.out.println(GREEN + "✓ Connected to " + BOLD + host + RESET);
                    
                    conn.close();
                    break; // Stop trying other hosts

                } catch (TimeoutException ex) {
                    long elapsed = System.currentTimeMillis() - start;
                    System.out.println(RED + "✗ TIMEOUT: Host did not respond within " + CONNECTION_TIMEOUT_SECONDS + " seconds" + RESET);
                    System.out.println(RED + "  (Possible: Firewall blocking, Host down, Port closed)" + RESET);
                    
                    if (conn != null) {
                        try { conn.close(); } catch (Exception e) {}
                    }
                    // Continue to next host

                } catch (SQLException ex) {
                    long elapsed = System.currentTimeMillis() - start;
                    System.out.println(RED + "✗ Failed: " + ex.getErrorCode() + " - " + ex.getMessage() + RESET);
                    
                    if (conn != null) {
                        try { conn.close(); } catch (Exception e) {}
                    }
                    
                    if (ex.getErrorCode() == 1017 || "28000".equals(ex.getSQLState())) {
                        System.out.println(RED + BOLD + "Authentication failed. Stopping." + RESET);
                        break;
                    }
                    // Continue to next host
                }
            }

            // Report results
            if (connectedHost != null) {
                String connectedIP = resolveIP(connectedHost);
                String backendIP = resolveIP(backendNode);

                System.out.println(GREEN + "RESULT: SUCCESS" + RESET);
                System.out.println(GREEN + "Connected successfully to service: " + BOLD + service + RESET);

                System.out.println(GREEN +
                        "Connected using SCAN host and IP: " +
                        BOLD + connectedHost + RESET +
                        " and " + BOLD + connectedIP + RESET);

                System.out.println(GREEN +
                        "Backend RAC node and IP: " +
                        BOLD + backendNode + RESET +
                        " and " + BOLD + backendIP + RESET);

                System.out.println(GREEN +
                        "Time taken: " + BOLD + (totalTime / 1000) +
                        " sec (" + totalTime + " ms)" + RESET);
            } else {
                System.out.println(RED + "RESULT: FAILURE" + RESET);
                System.out.println(RED + "Could not connect to any configured SCAN host" + RESET);
                for (String host : scanHosts) {
                    System.out.println(RED + "  - " + host + RESET);
                }
            }

            entryNumber++;
        }

        System.out.println(CYAN + "\n=== Testing Complete ===" + RESET);
    }
}
