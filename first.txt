SELECT 
    CASE 
        WHEN tok.TENANT_ID = -1234 THEN 'carbon.super'
        ELSE tnt.UM_DOMAIN_NAME 
    END AS BANK,
    app.ORGANIZATION,
    app.NAME AS APPLICATION_NAME,
    apps.CONSUMER_KEY,
    -- Grouping by the date portion only
    CAST(tok.TIME_CREATED AS DATE) AS CREATED_DATE,
    COUNT(tok.ACCESS_TOKEN) AS TOKENS_ISSUED_COUNT
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
JOIN 
    COCT42_AM.AM_APPLICATION_KEY_MAPPING map ON apps.CONSUMER_KEY = map.CONSUMER_KEY
JOIN 
    COCT42_AM.AM_APPLICATION app ON map.APPLICATION_ID = app.APPLICATION_ID
LEFT JOIN 
    COCT42_USER.UM_TENANT tnt ON tok.TENANT_ID = tnt.UM_ID
WHERE 
    -- Filter for the last 7 days
    tok.TIME_CREATED >= SYSTIMESTAMP - INTERVAL '7' DAY
    -- We usually look at all tokens created (ACTIVE or EXPIRED) to see the true trend
    AND tok.TOKEN_STATE IN ('ACTIVE', 'EXPIRED')
GROUP BY 
    tok.TENANT_ID,
    tnt.UM_DOMAIN_NAME, 
    app.ORGANIZATION, 
    app.NAME, 
    apps.CONSUMER_KEY,
    CAST(tok.TIME_CREATED AS DATE)
ORDER BY 
    CREATED_DATE DESC, 
    TOKENS_ISSUED_COUNT DESC;
