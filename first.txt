# Watch WSO2 thread count every 1 second
while true; do
  echo "$(date) - Threads: $(ps -eLf | grep wso2 | wc -l)"
  sleep 1
done

bash# Get WSO2 PID first
ps aux | grep wso2

# Then watch thread states every second (replace <PID>)
while true; do
  echo "=== $(date) ==="
  jstack <PID> | grep "state=" | sort | uniq -c | sort -rn
  sleep 2
done
