SELECT 
    -- If Tenant ID is -1234, label it 'carbon.super', otherwise use the table name
    CASE 
        WHEN tok.TENANT_ID = -1234 THEN 'carbon.super'
        ELSE tnt.UM_DOMAIN_NAME 
    END AS BANK,
    app.ORGANIZATION,
    app.NAME AS APPLICATION_NAME,
    apps.CONSUMER_KEY,
    COUNT(tok.ACCESS_TOKEN) AS TOTAL_COUNT
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
JOIN 
    COCT42_AM.AM_APPLICATION_KEY_MAPPING map ON apps.CONSUMER_KEY = map.CONSUMER_KEY
JOIN 
    COCT42_AM.AM_APPLICATION app ON map.APPLICATION_ID = app.APPLICATION_ID
-- Use LEFT JOIN so we don't lose the -1234 records
LEFT JOIN 
    COCT42_USER.UM_TENANT tnt ON tok.TENANT_ID = tnt.UM_ID
WHERE 
    tok.TOKEN_STATE = 'ACTIVE'
    AND (
        tok.VALIDITY_PERIOD < 0 
        OR 
        ( ( (CAST(SYS_EXTRACT_UTC(SYSTIMESTAMP) AS DATE) - TO_DATE('01-01-1970', 'DD-MM-YYYY')) * 86400000 )
          - 
          ( (CAST(tok.TIME_CREATED AS DATE) - TO_DATE('01-01-1970', 'DD-MM-YYYY')) * 86400000 )
        ) < tok.VALIDITY_PERIOD
    )
GROUP BY 
    tok.TENANT_ID,
    tnt.UM_DOMAIN_NAME, 
    app.ORGANIZATION, 
    app.NAME, 
    apps.CONSUMER_KEY
ORDER BY 
    TOTAL_COUNT DESC;
