SELECT 
    tnt.UM_DOMAIN_NAME AS BANK,
    app.ORGANIZATION,
    app.NAME AS APPLICATION_NAME,
    apps.CONSUMER_KEY,
    COUNT(tok.ACCESS_TOKEN) AS ACTIVE_TOKEN_COUNT
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps 
    ON tok.CONSUMER_KEY_ID = apps.ID
JOIN 
    COCT42_AM.AM_APPLICATION_KEY_MAPPING map 
    ON apps.CONSUMER_KEY = map.CONSUMER_KEY
JOIN 
    COCT42_AM.AM_APPLICATION app 
    ON map.APPLICATION_ID = app.APPLICATION_ID
JOIN 
    COCT42_USER.UM_TENANT tnt 
    ON tok.TENANT_ID = tnt.UM_ID
WHERE 
    tok.TOKEN_STATE = 'ACTIVE'
    -- 1. Ignore the 24 rows with negative/null validity
    AND tok.VALIDITY_PERIOD > 0 
    -- 2. Direct Timestamp comparison (No addition math on the column)
    -- This checks: Is (Now - Created) < Validity?
    AND (SYSTIMESTAMP - tok.TIME_CREATED) < NUMTODSINTERVAL(tok.VALIDITY_PERIOD/1000, 'SECOND')
GROUP BY 
    tnt.UM_DOMAIN_NAME, 
    app.ORGANIZATION, 
    app.NAME, 
    apps.CONSUMER_KEY
ORDER BY 
    ACTIVE_TOKEN_COUNT DESC;
