SELECT 
    tok.ACCESS_TOKEN,
    tok.TOKEN_STATE,
    tok.TIME_CREATED,
    -- Calculated Expiry Field
    (tok.TIME_CREATED + (tok.VALIDITY_PERIOD / 86400000)) AS CALCULATED_EXPIRY,
    -- Is it expired according to the clock?
    CASE 
        WHEN (tok.TIME_CREATED + (tok.VALIDITY_PERIOD / 86400000)) < SYSDATE THEN 'YES'
        ELSE 'NO'
    END AS IS_PAST_EXPIRY
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
WHERE 
    apps.CONSUMER_KEY = '11245'
ORDER BY 
    tok.TIME_CREATED DESC
FETCH FIRST 100 ROWS ONLY;
