SELECT 
    CASE 
        WHEN tok.TENANT_ID = -1234 THEN 'carbon.super'
        ELSE tnt.UM_DOMAIN_NAME 
    END AS BANK,
    app.NAME AS APPLICATION_NAME,
    -- Count tokens by age buckets
    COUNT(CASE WHEN tok.TIME_CREATED >= SYSTIMESTAMP - INTERVAL '1' DAY THEN 1 END) AS CREATED_LAST_24H,
    COUNT(CASE WHEN tok.TIME_CREATED >= SYSTIMESTAMP - INTERVAL '7' DAY 
               AND tok.TIME_CREATED < SYSTIMESTAMP - INTERVAL '1' DAY THEN 1 END) AS CREATED_LAST_7_DAYS,
    COUNT(CASE WHEN tok.TIME_CREATED >= SYSTIMESTAMP - INTERVAL '30' DAY 
               AND tok.TIME_CREATED < SYSTIMESTAMP - INTERVAL '7' DAY THEN 1 END) AS CREATED_LAST_MONTH,
    COUNT(CASE WHEN tok.TIME_CREATED < SYSTIMESTAMP - INTERVAL '30' DAY THEN 1 END) AS OLDER_THAN_30_DAYS,
    COUNT(*) AS TOTAL_ACTIVE_TOKENS
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
JOIN 
    COCT42_AM.AM_APPLICATION_KEY_MAPPING map ON apps.CONSUMER_KEY = map.CONSUMER_KEY
JOIN 
    COCT42_AM.AM_APPLICATION app ON map.APPLICATION_ID = app.APPLICATION_ID
LEFT JOIN 
    COCT42_USER.UM_TENANT tnt ON tok.TENANT_ID = tnt.UM_ID
WHERE 
    tok.TOKEN_STATE = 'ACTIVE'
    AND apps.CONSUMER_KEY = '11245'
GROUP BY 
    tok.TENANT_ID, tnt.UM_DOMAIN_NAME, app.NAME;
