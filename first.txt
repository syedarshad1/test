SELECT 
    tok.ACCESS_TOKEN,
    tok.TIME_CREATED AS CREATED_UTC,
    -- Add the 1 hour validity (3600000ms / 86400000ms per day)
    (tok.TIME_CREATED + (3600000 / 86400000)) AS EXPIRY_UTC,
    -- Status
    tok.TOKEN_STATE
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
WHERE 
    apps.CONSUMER_KEY = '11245'
ORDER BY 
    tok.TIME_CREATED DESC
FETCH FIRST 10 ROWS ONLY;
