SELECT 
    t.CONSUMER_KEY_ID,
    ck.CONSUMER_KEY,
    ck.APP_NAME,
    COUNT(*) AS active_token_count
FROM COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN t
JOIN COCT42_AM.IDN_OAUTH2_CONSUMER_APPS ck
    ON t.CONSUMER_KEY_ID = ck.ID
WHERE t.TOKEN_STATE = 'ACTIVE'
GROUP BY t.CONSUMER_KEY_ID, ck.CONSUMER_KEY, ck.APP_NAME
ORDER BY active_token_count DESC;
