SELECT 
    tnt.UM_DOMAIN_NAME AS BANK,
    app.ORGANIZATION,
    app.NAME AS APPLICATION_NAME,
    map.CONSUMER_KEY,
    COUNT(tok.ACCESS_TOKEN) AS ACTIVE_TOKEN_COUNT
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.AM_APPLICATION_KEY_MAPPING map ON tok.CONSUMER_KEY_ID = map.APPLICATION_ID -- Or join via IDN_OAUTH_CONSUMER_APPS if ID exists
JOIN 
    COCT42_AM.AM_APPLICATION app ON map.APPLICATION_ID = app.APPLICATION_ID
JOIN 
    COCT42_USER.UM_TENANT tnt ON tok.TENANT_ID = tnt.UM_ID
WHERE 
    tok.TOKEN_STATE = 'ACTIVE' 
    -- Filtering for tokens that have not yet reached their expiry timestamp
    AND (tok.TIME_CREATED + tok.VALIDITY_PERIOD) > (
        (sysdate - to_date('01-JAN-1970','DD-MON-YYYY')) * 86400000
    )
GROUP BY 
    tnt.UM_DOMAIN_NAME, 
    app.ORGANIZATION, 
    app.NAME, 
    map.CONSUMER_KEY
ORDER BY 
    ACTIVE_TOKEN_COUNT DESC;
