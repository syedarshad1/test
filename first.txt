import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.io.Console;
import java.util.*;
import java.util.regex.*;
import java.net.*;

public class OracleConnectionTest {

    private static final String GREEN  = "\u001B[32m";
    private static final String RED    = "\u001B[31m";
    private static final String YELLOW = "\u001B[33m";
    private static final String CYAN   = "\u001B[36m";
    private static final String RESET  = "\u001B[0m";
    private static final String BOLD   = "\u001B[1m";

    private static String extractServiceName(String jdbcUrl) {
        Pattern p = Pattern.compile("service_name=([^\\)]+)", Pattern.CASE_INSENSITIVE);
        Matcher m = p.matcher(jdbcUrl);
        return m.find() ? m.group(1).trim() : "UNKNOWN_SERVICE";
    }

    private static List<String> extractHosts(String jdbcUrl) {
        List<String> hosts = new ArrayList<>();
        Pattern p = Pattern.compile("host=([^\\)]+)", Pattern.CASE_INSENSITIVE);
        Matcher m = p.matcher(jdbcUrl);
        while (m.find()) hosts.add(m.group(1).trim());
        return hosts;
    }

    private static String extractPort(String jdbcUrl, String host) {
        // Find the port for the specific host
        Pattern p = Pattern.compile("host=" + Pattern.quote(host) + "\\)\\(protocol=tcp\\)\\(port=([0-9]+)\\)", Pattern.CASE_INSENSITIVE);
        Matcher m = p.matcher(jdbcUrl);
        if (m.find()) {
            return m.group(1);
        }
        // Fallback pattern
        p = Pattern.compile("\\(port=([0-9]+)\\)", Pattern.CASE_INSENSITIVE);
        m = p.matcher(jdbcUrl);
        if (m.find()) {
            return m.group(1);
        }
        return "1521";
    }

    private static String resolveIP(String host) {
        try {
            InetAddress addr = InetAddress.getByName(host);
            return addr.getHostAddress();
        } catch (Exception e) {
            return "DNS_LOOKUP_FAILED";
        }
    }

    private static String getActualConnectedHost(Connection conn, List<String> configuredHosts, String jdbcUrl) {
        try {
            // Query using SYS_CONTEXT to get server host
            java.sql.Statement stmt = conn.createStatement();
            java.sql.ResultSet rs = stmt.executeQuery(
                    "SELECT sys_context('USERENV','SERVER_HOST') AS server_host FROM dual");

            String serverHost = null;
            if (rs.next()) {
                serverHost = rs.getString("server_host");
            }
            rs.close();
            stmt.close();

            // Match the server host with configured hosts in order
            if (serverHost != null && !serverHost.trim().isEmpty()) {
                // Try exact match first
                for (String configHost : configuredHosts) {
                    if (configHost.toLowerCase().equals(serverHost.toLowerCase())) {
                        return configHost;
                    }
                }
                // Try partial match
                for (String configHost : configuredHosts) {
                    if (configHost.toLowerCase().contains(serverHost.toLowerCase()) ||
                        serverHost.toLowerCase().contains(configHost.toLowerCase())) {
                        return configHost;
                    }
                }
                // Return what we got from the query
                return serverHost;
            }
            
            // Fallback: return first host from JDBC string
            if (!configuredHosts.isEmpty()) {
                return configuredHosts.get(0);
            }
            
            return "UNKNOWN_HOST";
        } catch (SQLException e) {
            // Fallback: return first resolvable host
            for (String host : configuredHosts) {
                String ip = resolveIP(host);
                if (!ip.equals("DNS_LOOKUP_FAILED")) {
                    return host;
                }
            }
            // Last resort: return first configured host
            if (!configuredHosts.isEmpty()) {
                return configuredHosts.get(0);
            }
            return "UNABLE_TO_QUERY";
        }
    }

    public static void main(String[] args) {

        System.out.println(CYAN + "=== Oracle JDBC Diagnostic Tool ===" + RESET);

        String tnsPath = (args.length > 0)
                ? args[0]
                : Paths.get("tns.txt").toAbsolutePath().toString();

        System.out.println(CYAN + "Using file: " + tnsPath + RESET);

        List<String> lines;

        try {
            lines = Files.readAllLines(Paths.get(tnsPath));
        } catch (Exception e) {
            System.out.println(RED + "FAILED: Unable to read " + tnsPath + RESET);
            return;
        }

        lines.removeIf(line -> line.trim().isEmpty() || line.trim().startsWith("#"));

        if (lines.isEmpty()) {
            System.out.println(RED + "No JDBC URLs found." + RESET);
            return;
        }

        Console console = System.console();
        if (console == null) {
            System.out.println(RED + "No console available." + RESET);
            return;
        }

        int entryNumber = 1;
        int totalEntries = lines.size();

        // ---------------------------------------------------------
        // ONE LOOP — ASK CREDENTIALS → CONNECT → REPORT
        // ---------------------------------------------------------
        for (String url : lines) {

            String service = extractServiceName(url);

            System.out.println(CYAN + "\n==================================================" + RESET);
            System.out.println(CYAN + "Entry " + entryNumber + " of " + totalEntries + RESET);

            // Extract hosts from JDBC URL
            List<String> scanHosts = extractHosts(url);
            System.out.println(CYAN + "Configured hosts in order: " + scanHosts + RESET);

            // Ask credentials immediately
            System.out.println(CYAN + "--- Credentials for " + service + " ---" + RESET);
            String user = console.readLine("Enter username for " + service + ": ");
            char[] passChars = console.readPassword("Enter password for " + service + ": ");
            String pass = new String(passChars);

            long start = System.currentTimeMillis();

            try (Connection conn = DriverManager.getConnection(url, user, pass)) {

                long elapsedMs = System.currentTimeMillis() - start;
                long elapsedSec = elapsedMs / 1000;

                // Get the ACTUAL connected host
                String usedScanHost = getActualConnectedHost(conn, scanHosts, url);
                String usedScanIP = resolveIP(usedScanHost);

                // Get server host using SYS_CONTEXT
                String serverHost = "UNKNOWN_HOST";
                
                java.sql.Statement stmt = conn.createStatement();
                java.sql.ResultSet rs = stmt.executeQuery(
                        "SELECT sys_context('USERENV','SERVER_HOST') AS server_host FROM dual");

                if (rs.next()) {
                    serverHost = rs.getString("server_host");
                }

                rs.close();
                stmt.close();

                // Server host IP
                String serverHostIP = resolveIP(serverHost);

                System.out.println(GREEN + "RESULT: SUCCESS" + RESET);
                System.out.println(GREEN + "Connected successfully to service: " + BOLD + service + RESET);

                System.out.println(GREEN +
                        "Connected using SCAN host and IP: " +
                        BOLD + usedScanHost + RESET +
                        " and " + BOLD + usedScanIP + RESET);

                System.out.println(GREEN +
                        "Backend RAC node and IP: " +
                        BOLD + serverHost + RESET +
                        " and " + BOLD + serverHostIP + RESET);

                System.out.println(GREEN +
                        "Time taken: " + BOLD + elapsedSec +
                        " sec (" + elapsedMs + " ms)" + RESET);

            } catch (SQLException ex) {

                long elapsedMs = System.currentTimeMillis() - start;
                long elapsedSec = elapsedMs / 1000;

                System.out.println(RED + "RESULT: FAILURE" + RESET);
                System.out.println(RED + "Connection failed for descriptor:" + RESET);
                System.out.println(url);
                System.out.println(RED + "Time taken: " + elapsedSec + " sec (" + elapsedMs + " ms)" + RESET);
                System.out.println(RED + "SQLState : " + ex.getSQLState() + RESET);
                System.out.println(RED + "ErrorCode: " + ex.getErrorCode() + RESET);
                System.out.println(RED + "Message  : " + ex.getMessage() + RESET);

                if (ex.getErrorCode() == 1017 || "28000".equals(ex.getSQLState())) {
                    System.out.println(RED + BOLD +
                            "Authentication failed. Stopping further attempts." +
                            RESET);
                    break;
                }
            }

            entryNumber++;
        }

        System.out.println(CYAN + "\n=== Testing Complete ===" + RESET);
    }
}
