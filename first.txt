WITH ActiveTokenCounts AS (
    SELECT 
        tok.TENANT_ID, 
        tok.CONSUMER_KEY_ID, 
        COUNT(tok.ACCESS_TOKEN) AS TOKEN_COUNT
    FROM 
        COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
    WHERE 
        tok.TOKEN_STATE = 'ACTIVE'
        -- Ignore 'Zero' or invalid years (the source of ORA-01841)
        AND EXTRACT(YEAR FROM tok.TIME_CREATED) > 1970
        -- Comparison logic using simple date math
        AND (CAST(tok.TIME_CREATED AS DATE) + (tok.VALIDITY_PERIOD / 86400000)) > SYSDATE
    GROUP BY 
        tok.TENANT_ID, 
        tok.CONSUMER_KEY_ID
)
SELECT 
    tnt.UM_DOMAIN_NAME AS BANK,
    app.ORGANIZATION,
    app.NAME AS APPLICATION_NAME,
    apps.CONSUMER_KEY,
    res.TOKEN_COUNT
FROM 
    ActiveTokenCounts res
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON res.CONSUMER_KEY_ID = apps.ID
JOIN 
    COCT42_AM.AM_APPLICATION_KEY_MAPPING map ON apps.CONSUMER_KEY = map.CONSUMER_KEY
JOIN 
    COCT42_AM.AM_APPLICATION app ON map.APPLICATION_ID = app.APPLICATION_ID
JOIN 
    COCT42_USER.UM_TENANT tnt ON res.TENANT_ID = tnt.UM_ID
ORDER BY 
    res.TOKEN_COUNT DESC;
