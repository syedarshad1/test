SELECT 
    -- Adjusting UTC data to Local Time (-05:00)
    TO_CHAR(FROM_TZ(CAST(tok.TIME_CREATED AS TIMESTAMP), 'UTC') 
        AT TIME ZONE '-05:00', 'YYYY-MM-DD HH24') || ':00' AS LOCAL_HOURLY_WINDOW,
    COUNT(*) AS NEW_TOKENS_GENERATED
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
WHERE 
    apps.CONSUMER_KEY = '11245'
    AND tok.TIME_CREATED >= SYSTIMESTAMP - INTERVAL '4' HOUR
GROUP BY 
    TO_CHAR(FROM_TZ(CAST(tok.TIME_CREATED AS TIMESTAMP), 'UTC') 
        AT TIME ZONE '-05:00', 'YYYY-MM-DD HH24')
ORDER BY 
    LOCAL_HOURLY_WINDOW DESC;
