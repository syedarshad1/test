SELECT 
    BANK,
    ORGANIZATION,
    APPLICATION_NAME,
    CONSUMER_KEY,
    SUM(CASE WHEN days_ago = 0 THEN 1 ELSE 0 END) AS TODAY,
    SUM(CASE WHEN days_ago = 1 THEN 1 ELSE 0 END) AS YESTERDAY,
    SUM(CASE WHEN days_ago = 2 THEN 1 ELSE 0 END) AS DAY_3,
    SUM(CASE WHEN days_ago = 3 THEN 1 ELSE 0 END) AS DAY_4,
    SUM(CASE WHEN days_ago = 4 THEN 1 ELSE 0 END) AS DAY_5,
    SUM(CASE WHEN days_ago = 5 THEN 1 ELSE 0 END) AS DAY_6,
    SUM(CASE WHEN days_ago = 6 THEN 1 ELSE 0 END) AS DAY_7
FROM (
    SELECT 
        CASE 
            WHEN tok.TENANT_ID = -1234 THEN 'carbon.super'
            ELSE tnt.UM_DOMAIN_NAME 
        END AS BANK,
        app.ORGANIZATION,
        app.NAME AS APPLICATION_NAME,
        apps.CONSUMER_KEY,
        -- Calculate how many days ago the token was created
        TRUNC(CAST(SYSTIMESTAMP AS DATE)) - TRUNC(CAST(tok.TIME_CREATED AS DATE)) AS days_ago
    FROM 
        COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
    JOIN 
        COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
    JOIN 
        COCT42_AM.AM_APPLICATION_KEY_MAPPING map ON apps.CONSUMER_KEY = map.CONSUMER_KEY
    JOIN 
        COCT42_AM.AM_APPLICATION app ON map.APPLICATION_ID = app.APPLICATION_ID
    LEFT JOIN 
        COCT42_USER.UM_TENANT tnt ON tok.TENANT_ID = tnt.UM_ID
    WHERE 
        -- Look at everything created in the last 7 days
        tok.TIME_CREATED >= TRUNC(SYSTIMESTAMP) - 7
)
GROUP BY 
    BANK, 
    ORGANIZATION, 
    APPLICATION_NAME, 
    CONSUMER_KEY
ORDER BY 
    TODAY DESC;
