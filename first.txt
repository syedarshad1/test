SELECT 
    TO_CHAR(tok.TIME_CREATED, 'YYYY-MM-DD HH24') || ':00' AS HOURLY_WINDOW,
    COUNT(*) AS NEW_TOKENS_GENERATED
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
WHERE 
    apps.CONSUMER_KEY = '11245'
    -- Look back exactly 4 hours from right now
    AND tok.TIME_CREATED >= SYSTIMESTAMP - INTERVAL '4' HOUR
GROUP BY 
    TO_CHAR(tok.TIME_CREATED, 'YYYY-MM-DD HH24')
ORDER BY 
    HOURLY_WINDOW DESC;
