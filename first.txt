SELECT 
    CASE 
        WHEN tok.TENANT_ID = -1234 THEN 'carbon.super'
        ELSE tnt.UM_DOMAIN_NAME 
    END AS BANK,
    app.ORGANIZATION,
    app.NAME AS APPLICATION_NAME,
    apps.CONSUMER_KEY,
    -- 1. Current Active Tokens (Still valid and not expired)
    SUM(CASE WHEN tok.TOKEN_STATE = 'ACTIVE' 
             AND (tok.VALIDITY_PERIOD < 0 OR 
                 ((CAST(SYS_EXTRACT_UTC(SYSTIMESTAMP) AS DATE) - TO_DATE('01-01-1970','DD-MM-YYYY'))*86400000) - 
                 ((CAST(tok.TIME_CREATED AS DATE) - TO_DATE('01-01-1970','DD-MM-YYYY'))*86400000) < tok.VALIDITY_PERIOD)
        THEN 1 ELSE 0 END) AS CURRENT_ACTIVE_TOKENS,
    -- 2. Last 7 Days Trend (Generated tokens per day)
    SUM(CASE WHEN TRUNC(CAST(tok.TIME_CREATED AS DATE)) = TRUNC(SYSDATE) THEN 1 ELSE 0 END) AS TODAY,
    SUM(CASE WHEN TRUNC(CAST(tok.TIME_CREATED AS DATE)) = TRUNC(SYSDATE) - 1 THEN 1 ELSE 0 END) AS YESTERDAY,
    SUM(CASE WHEN TRUNC(CAST(tok.TIME_CREATED AS DATE)) = TRUNC(SYSDATE) - 2 THEN 1 ELSE 0 END) AS DAY_3,
    SUM(CASE WHEN TRUNC(CAST(tok.TIME_CREATED AS DATE)) = TRUNC(SYSDATE) - 3 THEN 1 ELSE 0 END) AS DAY_4,
    SUM(CASE WHEN TRUNC(CAST(tok.TIME_CREATED AS DATE)) = TRUNC(SYSDATE) - 4 THEN 1 ELSE 0 END) AS DAY_5,
    SUM(CASE WHEN TRUNC(CAST(tok.TIME_CREATED AS DATE)) = TRUNC(SYSDATE) - 5 THEN 1 ELSE 0 END) AS DAY_6,
    SUM(CASE WHEN TRUNC(CAST(tok.TIME_CREATED AS DATE)) = TRUNC(SYSDATE) - 6 THEN 1 ELSE 0 END) AS DAY_7
FROM 
    COCT42_AM.IDN_OAUTH2_ACCESS_TOKEN tok
JOIN 
    COCT42_AM.IDN_OAUTH_CONSUMER_APPS apps ON tok.CONSUMER_KEY_ID = apps.ID
JOIN 
    COCT42_AM.AM_APPLICATION_KEY_MAPPING map ON apps.CONSUMER_KEY = map.CONSUMER_KEY
JOIN 
    COCT42_AM.AM_APPLICATION app ON map.APPLICATION_ID = app.APPLICATION_ID
LEFT JOIN 
    COCT42_USER.UM_TENANT tnt ON tok.TENANT_ID = tnt.UM_ID
WHERE 
    -- Broad filter to capture current active + last 7 days history
    (tok.TOKEN_STATE = 'ACTIVE' OR tok.TIME_CREATED >= TRUNC(SYSTIMESTAMP) - 7)
    AND app.NAME = 'YOUR_APPLICATION_NAME'
GROUP BY 
    tok.TENANT_ID,
    tnt.UM_DOMAIN_NAME, 
    app.ORGANIZATION, 
    app.NAME, 
    apps.CONSUMER_KEY;
